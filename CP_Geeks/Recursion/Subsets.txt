class Solution{
public:
	void helper (vector<vector<int>>& ans, vector<int>& nums, vector<int>& current,int index){
		//base case
		if(index >= nums.size()){
			ans.push_back(current);
			return;
		}
		
		//take nums[i]
		current.push_back(nums[index]);
		helper(ans,nums,current,index+1);
		
		//backtrack
		current.pop_back();
		
		
		//don't take nums[i]
		
		helper(ans,nums,current,index+1);
		
		
	
	vector<vector<int>>subsets (vector<int>nums){
	vector<int>current;
	vector<vector<int>>ans;
	
	helper(ans,nums,current,0);
	
	return ans;
	}
}
